# 命名和可选参数
``` c#
Test(i: 1, j: 1.2f, str: "123");
Test1(i: 2, str: "3");//跳过其中的默认参数
void Test(int i, float j, string str)
{

}
void Test1(int i, float j = 1, string str = "2")
{

}
```

# 动态类型
``` c#
class Test1
{
    public void TestTest()
    {

    }
}
class Test2
{
    public void TestTest()
    {

    }
}
public class L3
{
    void Start()
    {
        dynamic dyn = 1;
        object obj = 2;
        print(obj.GetType());
        print(dyn.GetType());
        dyn += 1;
        print(dyn);//3

        //在确认对象成员 但是不确定类型时 使用
        object o = new Test1();
        (o as Test1).TestTest();
        dynamic d = o;
        d.TestTest();//避免反射/类型转换等
    }
}
```
